def make_window_data(signal_in, signal_out, window):
    """
    signal_in  -> compressed / distorted real-valued signal
    signal_out -> desired target / actual real-valued signal

    Returns:
      X : sliding windows from signal_in
      y : aligned target samples from signal_out
    """
    a = signal_in.ravel()
    y = signal_out[:len(a)].reshape(-1, 1)

    a_padded = np.pad(a, (window - 1, 0), constant_values=0)
    X = sliding_window_view(a_padded, window_shape=window, axis=0)

    return X[:len(y)], y[:len(X)]


def prepare_data(mat_path, window):
    sd = loadmat(mat_path)

    raw_in  = sd["train_input_real"].ravel()
    raw_out = sd["train_output_real"].ravel()

    X_raw, y_raw = make_window_data(raw_in, raw_out, window)

    N = len(X_raw)
    X_dev_raw,  y_dev_raw  = X_raw[:N // 2], y_raw[:N // 2]
    X_test_raw, y_test_raw = X_raw[N // 2:], y_raw[N // 2:]

    sv = int(0.8 * len(X_dev_raw))
    X_train_raw, y_train_raw = X_dev_raw[:sv], y_dev_raw[:sv]
    X_val_raw,   y_val_raw   = X_dev_raw[sv:], y_dev_raw[sv:]

    X_mean = X_train_raw.mean(0, keepdims=True)
    X_std  = X_train_raw.std(0, keepdims=True) + 1e-8
    y_mean = y_train_raw.mean(0, keepdims=True)
    y_std  = y_train_raw.std(0, keepdims=True) + 1e-8

    def sx(a): return (a - X_mean) / X_std
    def sy(a): return (a - y_mean) / y_std

    return {
        "X_train": sx(X_train_raw),
        "y_train": sy(y_train_raw),
        "X_val":   sx(X_val_raw),
        "y_val":   sy(y_val_raw),
        "X_test":  sx(X_test_raw),
        "y_test":  sy(y_test_raw),

        "X_train_raw": X_train_raw,
        "X_val_raw":   X_val_raw,
        "X_test_raw":  X_test_raw,
        "y_train_raw": y_train_raw,
        "y_val_raw":   y_val_raw,
        "y_test_raw":  y_test_raw,

        # current compressed sample aligned with each target sample
        "cmp_test": X_test_raw[:, -1].ravel(),

        "y_mean": y_mean,
        "y_std":  y_std,
    }


def unscale(y_s, d):
    return y_s * d["y_std"] + d["y_mean"]


def metrics(y_true, y_pred):
    yt = np.ravel(y_true)
    yp = np.ravel(y_pred)
    mse = mean_squared_error(yt, yp)
    return {
        "MSE":  mse,
        "RMSE": np.sqrt(mse),
        "MAE":  mean_absolute_error(yt, yp),
        "R2":   r2_score(yt, yp),
    }


def moving_average(x, win):
    win = max(1, int(win))
    kernel = np.ones(win) / win
    return np.convolve(x, kernel, mode="valid")


def analytic_envelope(x):
    """
    Envelope of a real signal via the Hilbert transform:
      A(t) = |hilbert(x(t))|
    """
    x = np.ravel(x)
    return np.abs(hilbert(x))


def envelope_relative_error(env_true, y_true, y_pred, floor_frac=0.01):
    """
    Relative error using raw signal residuals normalised by the true
    envelope amplitude:

        rel_err(t) = |y_pred(t) - y_true(t)| / max(A_true(t), ε)

    where  A_true = |hilbert(y_true)|  and  ε = floor_frac * max(A_true).

    This measures how large the pointwise prediction error is relative to
    the local signal amplitude, rather than comparing envelope estimates
    directly.
    """
    env_true = np.ravel(env_true)
    y_true   = np.ravel(y_true)
    y_pred   = np.ravel(y_pred)

    floor = floor_frac * np.max(env_true)
    denom = np.maximum(env_true, floor)

    rel_err = np.abs(y_pred - y_true) / denom
    return rel_err, floor


def find_segment_by_envelope(env, lo, hi, win=80, n=500):
    """
    Find a window whose mean envelope falls in [lo, hi).
    """
    step = max(1, len(env) // n)
    cands = [
        i for i in range(0, len(env) - win, step)
        if lo <= env[i:i + win].mean() < hi
    ]

    if not cands:
        all_means = [env[i:i + win].mean() for i in range(0, len(env) - win, step)]
        target = 0.5 * (lo + hi)
        cands = [np.argmin(np.abs(np.array(all_means) - target)) * step]

    return min(cands, key=lambda i: abs(env[i:i + win].mean() - 0.5 * (lo + hi)))

        **SHARED_PARAMS,
        hidden_layer_sizes=arch,
        random_state=42 + i
    )
    mlp.fit(data["X_train"], np.ravel(data["y_train"]))

    yhat_s = mlp.predict(data["X_test"]).reshape(-1, 1)
    yhat = np.ravel(unscale(yhat_s, data))

    ens_preds.append(yhat)
    loss_curves.append(mlp.loss_curve_)

yp  = np.mean(ens_preds, axis=0)                # ensemble prediction
xc  = np.ravel(data["cmp_test"])                # compressed input aligned to samples

# individual model metrics
ind_metrics = []

    im = metrics(yt, pred)
    ind_metrics.append({"arch": str(arch), **im})

        "mdape": np.median(vals) if len(vals) else np.nan,

    })

# ══════════════════════════════════════════════════════════════════════════════
# FIG B: SPECTROGRAM COMPARISON
# ══════════════════════════════════════════════════════════════════════════════
# Short-time Fourier transform (STFT) of the compressed input, actual target,
# and ensemble reconstruction — displayed as log-power spectrograms.
# A 4th panel shows the spectrogram of the residual (yt - yp) to directly
# reveal which time-frequency bins carry the most reconstruction error.
#
# Parameters — tune to your signal's characteristics:
#   FS      : sampling rate in Hz (set to 1 if unknown — axes show normalised freq)
#   NPERSEG : FFT window length (controls time vs freq resolution trade-off)
#   NOVERLAP: overlap between consecutive windows
# ══════════════════════════════════════════════════════════════════════════════

FS       = 1       # set to your actual sampling rate if known
NPERSEG  = 256
NOVERLAP = 224     # 87.5% overlap for smooth spectrogram

def compute_spec(signal, fs, nperseg, noverlap):
    f, t, Sxx = scipy_spectrogram(
        signal.ravel(), fs=fs,
        nperseg=nperseg, noverlap=noverlap,
        window="hann", scaling="spectrum"
    )
    Sxx_db = 10 * np.log10(Sxx + 1e-12)
    return f, t, Sxx_db


f_cmp,  t_cmp,  S_cmp  = compute_spec(xc,       FS, NPERSEG, NOVERLAP)
f_true, t_true, S_true = compute_spec(yt,       FS, NPERSEG, NOVERLAP)
f_pred, t_pred, S_pred = compute_spec(yp,       FS, NPERSEG, NOVERLAP)
f_res,  t_res,  S_res  = compute_spec(yt - yp,  FS, NPERSEG, NOVERLAP)

# Shared colour scale for the first three panels (input / target / prediction)
vmin_sig = np.percentile(S_true, 5)
vmax_sig = np.percentile(S_true, 99)

# Separate scale for residual (usually much smaller dynamic range)
vmin_res = np.percentile(S_res, 5)
vmax_res = np.percentile(S_res, 99)

freq_label = "Frequency (Hz)" if FS != 1 else "Normalised frequency (cycles/sample)"

fig_b = plt.figure(figsize=(PANEL_2C, PANEL_2C * 0.85))
gs_b  = gridspec.GridSpec(2, 2, figure=fig_b, hspace=0.42, wspace=0.35)

spec_panels = [
    (gs_b[0, 0], S_cmp,  t_cmp,  f_cmp,  "(a) Compressed input",   "Blues_r",  vmin_sig, vmax_sig),
    (gs_b[0, 1], S_true, t_true, f_true, "(b) Actual target",       "Blues_r",  vmin_sig, vmax_sig),
    (gs_b[1, 0], S_pred, t_pred, f_pred, "(c) Reconstruction",      "Blues_r",  vmin_sig, vmax_sig),
    (gs_b[1, 1], S_res,  t_res,  f_res,  "(d) Residual  y − ŷ",    "Oranges",  vmin_res, vmax_res),
]

for spec, Sxx, t_ax, f_ax, title, cmap, vlo, vhi in spec_panels:
    ax = fig_b.add_subplot(spec)
    im = ax.pcolormesh(
        t_ax, f_ax, Sxx,
        cmap=cmap, vmin=vlo, vmax=vhi,
        shading="gouraud", rasterized=True
    )
    cb = fig_b.colorbar(im, ax=ax, pad=0.02)
    cb.set_label("dB", fontsize=7)
    cb.ax.tick_params(labelsize=6)
    ax.set_xlabel("Time (samples)" if FS == 1 else "Time (s)")
    ax.set_ylabel(freq_label)
    ax.set_title(title)
    ax.xaxis.set_minor_locator(AutoMinorLocator())
    ax.yaxis.set_minor_locator(AutoMinorLocator())

fig_b.suptitle(
    "Fig. B — Spectrogram comparison: compressed input, actual target, "
    "reconstruction, and residual\n"
    f"(STFT: window = {NPERSEG} samples, overlap = {NOVERLAP} samples, "
    f"Hann window)",
    fontsize=9, fontweight="bold", y=1.02
)
